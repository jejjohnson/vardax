    Utilities & Diagnostics¶

    Supporting cast: a reference dynamical system for experiments, the statistical gates that decide whether an inference setup can be trusted, and the plotting helpers used throughout the end-to-end examples.

    Dynamical systems¶

    Lorenz-63 and Lorenz-96 are the standard chaotic testbeds for assimilation experiments; the simulate_* helpers generate trajectories for the Lorenz examples and the test suite, and time_patches produces the consecutive time pairs consumed by the one-step increment losses.

    vardax — Modular variational data assimilation with learned components.

    All public symbols are re-exported from the private _src subpackage so that user code imports from the top-level namespace:

    import vardax

model = vardax.FourDVarNet1D(...)

    class Lorenz63(eqx.Module):
    r"""Lorenz-63 vector field as an Equinox module.

    Implements the system

    $$
    \dot x = \sigma (y - x), \quad
    \dot y = x (\rho - z) - y, \quad
    \dot z = x y - \beta z .
    $$

    The module is a diffrax-compatible right-hand side ``f(t, y, args)``.

    Attributes:
        sigma: Prandtl number $\sigma$ (classically 10).
        rho: Rayleigh number $\rho$ (classically 28).
        beta: Geometric factor $\beta$ (classically 8/3).

    Examples:
        The vector field at $(1, 1, 1)$ for the classical parameters:

        >>> import jax.numpy as jnp
        >>> import vardax as vdx
        >>> f = vdx.Lorenz63(sigma=10.0, rho=28.0, beta=8.0 / 3.0)
        >>> dydt = f(0.0, jnp.ones(3), None)
        >>> (
        ...     dydt.shape,
        ...     bool(jnp.allclose(dydt, jnp.array([0.0, 26.0, 1.0 - 8.0 / 3.0]))),
        ... )
        ((3,), True)

        The origin is a fixed point:

        >>> bool(jnp.all(f(0.0, jnp.zeros(3), None) == 0))
        True

        It drives a [`DynTrajectory`][vardax.DynTrajectory] prior:

        >>> ts = jnp.linspace(0.0, 0.05, 6)
        >>> vdx.DynTrajectory(model=f)(jnp.ones(3), ts).shape
        (6, 3)
    """

    sigma: float
    rho: float
    beta: float

    def __call__(
        self,
        t: float | Float[Array, ""],
        y: Float[Array, " 3"],
        args: None,
    ) -> Float[Array, " 3"]:
        r"""Evaluate the vector field at state ``y`` and time ``t``.

        Args:
            t: Time (unused; the system is autonomous).
            y: State $(x, y, z)$.
            args: Unused, present for the diffrax signature.

        Returns:
            The time derivative $(\dot x, \dot y, \dot z)$.
        """
        state_x, state_y, state_z = y[0], y[1], y[2]
        # dx/dt = σ(y − x)
        dx_dt = self.sigma * (state_y - state_x)
        # dy/dt = x(ρ − z) − y
        dy_dt = state_x * (self.rho - state_z) - state_y
        # dz/dt = xy − βz
        dz_dt = state_x * state_y - self.beta * state_z
        return jnp.array([dx_dt, dy_dt, dz_dt])

    def get_config(self) -> dict[str, Any]:
        return {
            # float(): the parameters are leaves and may be 0-d arrays
            "sigma": float(self.sigma),
            "rho": float(self.rho),
            "beta": float(self.beta),
        }

    @classmethod
    def from_config(cls, config: dict[str, Any]) -> Lorenz63:
        return cls(**config)

    class Lorenz96(eqx.Module):
    r"""Lorenz-96 vector field as an Equinox module.

    Implements the system (with periodic boundary conditions)

    $$
    \dot x_k = (x_{k+1} - x_{k-2})\, x_{k-1} - x_k + F .
    $$

    The module is a diffrax-compatible right-hand side ``f(t, y, args)``.

    Attributes:
        forcing: Forcing constant $F$ (classically 8).

    Examples:
        The uniform state $x_k = F$ is a fixed point, and at the origin the
        field equals the forcing:

        >>> import jax.numpy as jnp
        >>> import vardax as vdx
        >>> f = vdx.Lorenz96(forcing=8.0)
        >>> bool(jnp.allclose(f(0.0, jnp.full(6, 8.0), None), 0.0))
        True
        >>> bool(jnp.allclose(f(0.0, jnp.zeros(6), None), 8.0))
        True

        It maps an $N$-vector to an $N$-vector, for any $N$:

        >>> f(0.0, jnp.arange(5.0), None).shape
        (5,)

        It drives a [`DynTrajectory`][vardax.DynTrajectory] prior:

        >>> ts = jnp.linspace(0.0, 0.05, 6)
        >>> vdx.DynTrajectory(model=f)(jnp.arange(5.0), ts).shape
        (6, 5)
    """

    forcing: float

    def __call__(
        self,
        t: float | Float[Array, ""],
        y: Float[Array, " N"],
        args: None,
    ) -> Float[Array, " N"]:
        r"""Evaluate the vector field at state ``y`` and time ``t``.

        Args:
            t: Time (unused; the system is autonomous).
            y: State $x$ on the periodic ring.
            args: Unused, present for the diffrax signature.

        Returns:
            The time derivative $\dot x$.
        """
        # Periodic shifts using jnp.roll
        y_km2 = jnp.roll(y, 2)  # x_{k-2}
        y_km1 = jnp.roll(y, 1)  # x_{k-1}
        y_kp1 = jnp.roll(y, -1)  # x_{k+1}
        return (y_kp1 - y_km2) * y_km1 - y + self.forcing

    def get_config(self) -> dict[str, Any]:
        return {
            # float(): the forcing is a leaf and may be a 0-d array
            "forcing": float(self.forcing),
        }

    @classmethod
    def from_config(cls, config: dict[str, Any]) -> Lorenz96:
        return cls(**config)

    def simulate_lorenz63(
    key: Array,
    *,
    sigma: float = 10.0,
    rho: float = 28.0,
    beta: float = 8.0 / 3.0,
    dt: float = 0.01,
    n_steps: int = 5000,
    n_burn_in: int = 1000,
    x0: Float[Array, " 3"] | None = None,
) -> tuple[Float[Array, " T"], Float[Array, "T 3"]]:
    r"""Simulate the Lorenz-63 system and return state trajectory.

    Integrates $\dot x = \sigma (y - x)$, $\dot y = x (\rho - z) - y$,
    $\dot z = x y - \beta z$ with diffrax and discards a burn-in so the
    trajectory lies on the attractor.

    Algorithm:
        1. Build the initial condition, $(1, 1, 1)$ perturbed by ``key``, or
           ``x0``.
        2. Integrate with ``Tsit5`` over ``n_burn_in + n_steps`` steps of size
           ``dt``, saving every step.
        3. Drop the first ``n_burn_in`` states and shift time to start at 0.

    Args:
        key: JAX PRNG key used to perturb the initial condition when
            ``x0`` is ``None``.
        sigma: Prandtl number $\sigma$.
        rho: Rayleigh number $\rho$.
        beta: Geometric factor $\beta$.
        dt: Integration time step.
        n_steps: Total number of integration steps *after* burn-in.
        n_burn_in: Number of initial steps to discard (burn-in).
        x0: Optional explicit initial condition.  When
            ``None``, the classic off-attractor point ``(1, 1, 1)`` is used,
            perturbed by ``key``. From there the transient onto the
            attractor takes a couple of time units, so the default burn-in
            lands on it. (Starting at the unstable fixed point instead, as
            older versions did, leaves the trajectory within a fraction of
            a unit of it for ~40 time units, far longer than any burn-in
            the tutorials use.)

    Returns:
        ``(time_coords, states)``: the time coordinates (starting at 0) and the
        Lorenz-63 state trajectory, with ``n_steps + 1`` rows.

    Examples:
        A short run without burn-in:

        >>> import jax
        >>> import vardax as vdx
        >>> t, states = vdx.simulate_lorenz63(
        ...     jax.random.PRNGKey(0), n_steps=10, n_burn_in=0
        ... )
        >>> t.shape, states.shape
        ((11,), (11, 3))
        >>> float(t[0])
        0.0

        An explicit initial condition is used as given:

        >>> import jax.numpy as jnp
        >>> x0 = jnp.array([1.0, 2.0, 3.0])
        >>> _, states = vdx.simulate_lorenz63(
        ...     jax.random.PRNGKey(0), n_steps=5, n_burn_in=0, x0=x0
        ... )
        >>> bool(jnp.allclose(states[0], x0))
        True

        A burn-in is dropped from the output but the time axis restarts at 0:

        >>> t, states = vdx.simulate_lorenz63(
        ...     jax.random.PRNGKey(0), n_steps=5, n_burn_in=5
        ... )
        >>> states.shape, float(t[0])
        ((6, 3), 0.0)
    """
    model = Lorenz63(sigma=sigma, rho=rho, beta=beta)

    if x0 is None:
        noise = jax.random.normal(key, shape=(3,)) * 0.01
        x0 = jnp.array([1.0, 1.0, 1.0]) + noise

    total_steps = n_burn_in + n_steps
    start_time = 0.0
    end_time = total_steps * dt

    save_times = jnp.linspace(start_time, end_time, total_steps + 1)

    solution = diffeqsolve(
        ODETerm(model),  # type: ignore[arg-type]  # ty:ignore[invalid-argument-type]
        Tsit5(),
        t0=start_time,
        t1=end_time,
        dt0=dt,
        y0=x0,
        saveat=SaveAt(ts=save_times),
        # Fixed-step integration takes exactly ``total_steps`` steps; diffrax's
        # default cap (4096) would abort long simulations (e.g. 5000 + 1000).
        max_steps=total_steps + 1,
    )

    # Discard burn-in steps; keep indices n_burn_in .. total_steps (inclusive)
    states = solution.ys[n_burn_in:]
    time_coords = save_times[n_burn_in:] - save_times[n_burn_in]
    return time_coords, states

    ForwardModel adapter that softly confines states to a box.

    Explicit Runge–Kutta steps of dynamics with quadratic terms (Lorenz-63, Lorenz-96, …) overflow, or drive an adaptive step-size controller past its step cap, when the state is far outside the attractor. A line-search minimiser such as the BFGS inside StrongFourDVar will try such states, and one non-finite cost is enough to poison the whole solve. This adapter maps the state through

    \[ s(x) = \operatorname{sign}(x)\,\begin{cases} |x| & |x| \le b \\ b + b\,\dfrac{u}{1 + u}, \quad u = (|x| - b)/b & |x| > b \end{cases} \]

    on the way into and out of every step, so that

    • states inside the box \(|x_i| \le b\) are untouched (choose bound so the attractor sits well inside it — the adapter is then exactly the wrapped model wherever the analysis can end up);
    • states outside are confined to \(|x_i| < 2b\), keeping every trial cost finite;
    • the map is continuously differentiable (slope one at the box edge from both sides) and its gradient outside, \(1/(1+u)^2\), stays representable in single precision for trial states up to about \(10^{19}\) times the bound, unlike a hard clip (zero gradient) or a tanh saturation (gradient underflows to zero a few bounds out); a rejected trial therefore still carries curvature information if the minimiser asks for it.
    Algorithm
    1. Confine the incoming state with \(s\).
    2. Advance it with the wrapped model's step.
    3. Confine the result with \(s\) again.

    r"""``ForwardModel`` adapter that softly confines states to a box.

    Explicit Runge–Kutta steps of dynamics with quadratic terms (Lorenz-63,
    Lorenz-96, …) overflow, or drive an adaptive step-size controller past
    its step cap, when the state is far outside the attractor. A
    line-search minimiser such as the BFGS inside
    [`StrongFourDVar`][vardax.StrongFourDVar] will try such states, and one
    non-finite cost is enough to poison the whole solve. This adapter maps
    the state through

    $$
    s(x) = \operatorname{sign}(x)\,\begin{cases}
      |x| & |x| \le b \\
      b + b\,\dfrac{u}{1 + u}, \quad u = (|x| - b)/b & |x| > b
    \end{cases}
    $$

    on the way into and out of every step, so that

    - states inside the box $|x_i| \le b$ are untouched (choose ``bound``
      so the attractor sits well inside it — the adapter is then exactly
      the wrapped model wherever the analysis can end up);
    - states outside are confined to $|x_i| < 2b$, keeping every trial
      cost finite;
    - the map is continuously differentiable (slope one at the box edge
      from both sides) and its gradient outside, $1/(1+u)^2$, stays
      representable in single precision for trial states up to about
      $10^{19}$ times the bound, unlike a hard clip (zero gradient) or a
      ``tanh`` saturation (gradient underflows to zero a few bounds out);
      a rejected trial therefore still carries curvature information if
      the minimiser asks for it.

    Algorithm:
        1. Confine the incoming state with $s$.
        2. Advance it with the wrapped model's ``step``.
        3. Confine the result with $s$ again.

    Attributes:
        forward: The wrapped ``pipekit_cycle.ForwardModel``.
        bound: Half-width $b$ of the box on which the adapter is the
            identity.

    Examples:
        >>> import jax.numpy as jnp
        >>> import vardax
        >>> class Double:
        ...     dt = 1.0
        ...     state_signature = None
        ...
        ...     def step(self, x, dt):
        ...         return 2.0 * x
        >>> fwd = vardax.SoftBoundedForward(Double(), bound=10.0)
        >>> fwd.step(jnp.array([1.0, -3.0]), 1.0)  # inside the box: untouched
        Array([ 2., -6.], dtype=float32)
        >>> bool(jnp.all(jnp.abs(fwd.step(jnp.array([1e6, -1e6]), 1.0)) < 20.0))
        True

        The step length and signature are forwarded from the wrapped model:

        >>> fwd.dt, fwd.state_signature
        (1.0, None)

        Wrapped around a diffrax-backed model it keeps a far-off state finite:

        >>> def decay(t, y, args):
        ...     return -y
        >>> inner = vardax.DynTrajectory(model=decay).as_forward_model(0.1)
        >>> safe = vardax.SoftBoundedForward(inner, bound=5.0)
        >>> bool(jnp.all(jnp.isfinite(safe.step(jnp.array([1e9]), 0.1))))
        True
    """

    forward: Any
    bound: float = eqx.field(static=True)

    @property
    def dt(self) -> float:
        return self.forward.dt

    @property
    def state_signature(self) -> Any:
        return getattr(self.forward, "state_signature", None)

    def _confine(self, state: Array) -> Array:
        half_width = self.bound
        abs_state = jnp.abs(state)
        # The in-box branch returns ``state`` itself (not ``sign * abs``) so the
        # gradient is exactly one everywhere inside, including at zero, where
        # ``jnp.sign`` would otherwise zero it. Outside, the algebraic
        # saturation u / (1 + u) keeps the derivative 1 / (1 + u)^2 finite and
        # non-zero far beyond any state a line search will try (tanh would
        # underflow to a zero gradient a few bounds out).
        # u = max(|x| − b, 0) / b
        excess = jnp.maximum(abs_state - half_width, 0.0) / half_width
        # s(x) = sign(x) (b + b u / (1 + u))
        outside = jnp.sign(state) * (half_width + half_width * excess / (1.0 + excess))
        return jnp.where(abs_state <= half_width, state, outside)

    def step(self, state: Array, dt: float) -> Array:
        r"""Advance ``state`` by ``dt`` with confinement before and after.

        Args:
            state: State $x$ to advance.
            dt: Step length $\Delta t$.

        Returns:
            $s(\Phi_{\Delta t}(s(x)))$, with $\Phi$ the wrapped model's step.
        """
        return self._confine(self.forward.step(self._confine(state), dt))

        x_true = sample_prior(key_prior)
        observation = simulate_obs(x_true, key_obs)

        scalar_true = float(jnp.linalg.norm(x_true))
        # ‖xᵢ‖₂ over every state dimension of each sample
        squared_norms = einx.sum("n [s...]", samples**2)
        scalar_samples = jnp.sqrt(squared_norms)
        rank = int(jnp.sum(scalar_samples < scalar_true))
        ranks.append(rank)

    p_oracle: Posterior,

    tolerance_sigma: float = 1.0,

    r"""Check that ``p_fast.mean`` lies within ``tolerance_sigma`` standard
    deviations of ``p_oracle.mean``.

    Marginal-only test: each component of the mean must satisfy

    $$
    \frac{|x^*_\text{fast} - x^*_\text{oracle}|}{\sigma_\text{post, oracle}}
    \le \text{tolerance\_sigma}.
    $$

    The oracle marginal $\sigma_i$ is extracted from
    ``p_oracle.cov`` by probing $e_i^T \Sigma e_i$ via one matvec per
    component. Cheap for moderate state size; for very large state

    directly via ``Posterior.cov`` materialised as a diagonal operator.

    Algorithm:
        1. Extract the oracle marginal standard deviations $\sigma_i$ from
           ``p_oracle.cov``.
        2. Form the mean difference $\Delta x^* = x^*_\text{fast} - x^*_\text{oracle}$.
        3. Compute the z-scores $z_i = |\Delta x^*_i| / \sigma_i$.
        4. Fail if any $z_i$ exceeds ``tolerance_sigma``.

    Args:
        p_fast: Posterior produced by the amortized / fast model.
        p_oracle: Posterior produced by the oracle (e.g. ``StrongFourDVar``
            + [`LaplaceCovariance`][vardax.LaplaceCovariance]).
        tolerance_sigma: Allowed deviation in units of $\sigma$.

    Raises:
        MissingInputError: If ``p_oracle.cov`` is ``None``.
        ShapeError: If the two means have different shapes.
        AssertionError: If any component exceeds the tolerance.

        A fast mean within one oracle standard deviation passes silently:

        >>> import jax
        >>> import jax.numpy as jnp
        >>> import lineax as lx
        >>> import vardax as vdx
        >>> cov = lx.IdentityLinearOperator(jax.ShapeDtypeStruct((3,), jnp.float32))

        >>> vdx.assert_posterior_agreement(fast, oracle, tolerance_sigma=1.0)

        >>> far = vdx.Posterior(mean=jnp.array([3.0, 0.0, 0.0]))
        >>> vdx.assert_posterior_agreement(
        ...     far, oracle
        ... )  # doctest: +IGNORE_EXCEPTION_DETAIL

        ...
        AssertionError: posterior agreement failed: max z = 3.000 > 1.000.

        An oracle without a covariance cannot supply the scale:

        >>> vdx.assert_posterior_agreement(
        ...     fast, fast
        ... )  # doctest: +IGNORE_EXCEPTION_DETAIL
        Traceback (most recent call last):
        ...
        MissingInputError: assert_posterior_agreement requires p_oracle.cov ...
    """

        raise MissingInputError(
            "assert_posterior_agreement requires p_oracle.cov to be set "
            "(used to extract the marginal standard deviations)."
        )
    if p_fast.mean.shape != p_oracle.mean.shape:
        raise ShapeError(
            "p_fast.mean and p_oracle.mean must have the same shape; got "
            f"{p_fast.mean.shape} vs {p_oracle.mean.shape}."
        )

    # z = |x*_fast − x*_oracle| / σ_oracle

    if jnp.any(z_score > tolerance_sigma):
        worst = float(jnp.max(z_score))
        raise AssertionError(
            f"posterior agreement failed: max z = {worst:.3f} > {tolerance_sigma:.3f}."
        )

    fn_fast: Callable[[Array], Array],
    fn_oracle: Callable[[Array], Array],
    y: Float[Array, ...],
    *,
    key: PRNGKeyArray,

    r"""Random-vector probe of the Jacobian agreement at ``y``.

         {\|J_\text{oracle} v\|} < \text{threshold}
    $$

    materialising either Jacobian — uses ``jax.jvp`` to apply each as
    needed. Cheaper than dense Jacobian comparison and is the
    operational test used by the six-step cycle.

    Algorithm:
        1. Draw a random unit vector $v$ of the shape of ``y``.
        2. Apply both Jacobians to it with ``jax.jvp``.
        3. Compute the relative error of the two products.
        4. Repeat for ``n_probes`` vectors and fail if the worst error exceeds
           ``threshold``.

    Args:
        fn_fast: Callable ``y → analysis`` (e.g.
            ``lambda y_: amortized(Batch1D(input=y_, mask=mask, target=None))``).
        fn_oracle: Same signature, but the oracle.
        y: Observation tensor at which the Jacobians are probed.
        key: PRNG key for the probe vectors.
        threshold: Maximum allowed relative error.
        n_probes: Number of probe vectors.

    Raises:
        AssertionError: If any probe exceeds the threshold.

    Examples:
        Two maps with the same Jacobian agree, even if their values differ:

        >>> import jax
        >>> import jax.numpy as jnp
        >>> import vardax as vdx
        >>> y = jnp.arange(4.0)
        >>> vdx.assert_adjoint_calibrated(
        ...     lambda v: 2.0 * v + 1.0,
        ...     lambda v: 2.0 * v,
        ...     y,
        ...     key=jax.random.PRNGKey(0),
        ... )

        >>> vdx.assert_adjoint_calibrated(
        ...     lambda v: 2.0 * v, lambda v: 3.0 * v, y, key=jax.random.PRNGKey(0)
        ... )  # doctest: +IGNORE_EXCEPTION_DETAIL
        Traceback (most recent call last):
        ...
        AssertionError: adjoint calibration failed: ...
    """
    keys = jax.random.split(key, n_probes)
    worst_rel = 0.0
    for probe_key in keys:
        probe = jax.random.normal(probe_key, y.shape)
        probe = probe / jnp.linalg.norm(probe)
        _, jvp_fast = jax.jvp(fn_fast, (y,), (probe,))
        _, jvp_oracle = jax.jvp(fn_oracle, (y,), (probe,))

        raise AssertionError(
            f"adjoint calibration failed: max relative error {worst_rel:.4f} > "
            f"{threshold:.4f} (over {n_probes} probes)."
        )

        sample_idx: Which batch element to visualize.

    Returns:
        ``(fig, axes)``: the Matplotlib figure and the array of three axes.

    Examples:
        >>> import matplotlib
        >>> matplotlib.use("Agg")
        >>> import numpy as np
        >>> import vardax as vdx
        >>> truth = np.random.default_rng(0).normal(size=(2, 6, 4))
        >>> fig, axes = vdx.plot_reconstruction_comparison(truth, truth, truth)
        >>> axes.shape, type(axes[0]).__name__
        ((3,), 'Axes')

        Another batch element can be selected:

        >>> fig, axes = vdx.plot_reconstruction_comparison(
        ...     truth, truth, truth, sample_idx=1
        ... )
        >>> axes[1].get_title()
        'Masked input'
    """
    import matplotlib.pyplot as plt

    fig, axes = plt.subplots(1, 3, figsize=(12, 4))
    data = [
        (target[sample_idx], "Target"),
        (masked_input[sample_idx], "Masked input"),
        (reconstruction[sample_idx], "Reconstruction"),
    ]
    for ax, (arr, title) in zip(axes, data, strict=False):
        ax.imshow(arr.T, aspect="auto", origin="lower")
        ax.set_title(title)
        ax.set_xlabel("time")
        ax.set_ylabel("feature")
    fig.tight_layout()
    return fig, axes
